    Notice
    \begin{align*}
        (1+x)\left( 1-x+x^2-x^3+\ldots +(-1)^{r-1}x^{r-1} \right)

\begin{exercise}
    For any prime ideal $\mathfrak{p}\subseteq A$, consider the map
    $$
    A\longrightarrow A / \mathfrak{p}
    $$

    From the problem we know $\forall x\in A,\exists n>0,\mathrm{s.t.}\ x^{n}=x$.
    So $x(x^{n-1}-1)=0$, and we obtain
    $$
    \overline{x}(\overline{x}^{n-1}-\overline{1})=0
    $$

    We may assume $x\not\in \mathfrak{p}$, i.e. $\overline{x}\neq 0$.
    Since $\mathfrak{p}$ is prime, $A / \mathfrak{p}$ is a integral domain.
    So $\overline{x}^{n-1}=\overline{1}$, $\overline{x}$ has its inverse in $A / \mathfrak{p}$.
    Because $x$ was an arbitary element in $A \backslash \mathfrak{p}$, $A / \mathfrak{p}$ is a field.
    Hence $\mathfrak{p}$ is maximal.
    \qed
\end{exercise}

\begin{exercise}
    \textbf{(POOR-WRITTEN UNCLEAR PROOF)}

    We prove by contradiction: Suppose $\mathfrak{a}=(1)$, then $\exists \text{polynomial }F\in A[f(x_{f})]$,
    for all $f$ and $x_{f}$,
    such that
    $$
    F(f(x_{f}),\ldots )=1,\text{ finitely many }f\in \Sigma
    $$
    For those $F$, we pick out one with the least number of $f\in \Sigma$ being involved.

    For a specific $f_0\in \Sigma$ which was involved in the polynomial $F$ above,
    consider the terms of $f_0$ with its highest degree $\alpha_0$ in $F(f(x_{f}),\ldots )$.
    Because $F(f(x_{f}),\ldots )=1$, those terms with degree $\alpha_0$ sums up to $0$, i.e.
    $$
    f(x_{f})^{\alpha_0}(\text{rest of the terms})=0
    $$
    Apparently $f(x_{f})^{\alpha_0}\neq 0$, so the rest of the terms sums up to $0$.
    Then for the rest of the terms, we can again similarly pick out a $f_1,\alpha_1$ and sort out "rest of the terms" to be $0$.
    We iterate the process above.
    Since there're only finitely many terms, the process must stop, making a contradiction.

    \emph{(We don't need to worry about characteristic of $K$ is finite or not, since w.l.o.g. we can let those terms in $F$ which adds up over the characteristic vanish in the first place.)}
\end{exercise}

            \exists f_{k_1},\ldots ,f_{k_{n}},
            \mathrm{s.t.}\ \bigcap_{i =1} ^{n}V(f_{k})\subseteq \bigcap_{f \in \mathcal{F}} V(f)
            $$
            $\Leftrightarrow$ 
            $$
            \forall f_1,f_2,\ldots \in A,
            \mathrm{s.t.}\ \mathcal{F}\subseteq (f_1,f_2,\ldots ),
            \exists f_{k_1},\ldots ,f_{k_{n}},
            \mathrm{s.t.}\ \mathcal{F}\subseteq (f_{k_1},\ldots ,f_{k_{n}})
            $$
            \begin{enumerate}[label=({\roman*})]
                \item $(\Rightarrow):$
                    We prove by contradiction and suppose $\mathcal{F}$ can only be infinite,
                    then let $f_1,f_2\ldots $ be the elements in $\mathcal{F}$, and such $f_{k_1},\ldots ,f_{k_{n}}$ cannot exist.
                \item $(\Leftarrow):$ 
                    When $\mathcal{F}$ is finite, it is a direct proposition from  \emph{(vi)}.
            \end{enumerate}
    \end{enumerate}
    \qed
\end{exercise}
