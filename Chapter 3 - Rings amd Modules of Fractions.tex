\begin{exercise}
    By \emph{Proposition 1.9},
    the problem is equivalent to prove that for any $\frac{a}{s}\in S^{-1}A$, we have
    $$
    \forall \frac{b}{t}\in S^{-1}A,1-\frac{a}{s}\frac{b}{t}\text{is a unit in }S^{-1}A
    $$
    Since $1+\mathfrak{a}$ is a multiplicative set, $\exists c\in \mathfrak{a},\text{s.t. }st=1+c$.
    Now
    \begin{align*}
        1-\frac{a}{s}\frac{b}{t}&= 1-\frac{ab}{st} \\
        &= 1-\frac{ab}{1+c} \\
        &= \frac{1+c-ab}{1+c}
    \end{align*}
    And we can easily see $\frac{1+c}{1+c-ab}=1+\frac{ab}{1+c-ab}\in S^{-1}A$.
    So the inverse of $1-\frac{a}{s}\frac{b}{t}$ is $1+\frac{ab}{1+c-ab}$.
    \qed
\end{exercise}

                    We'll prove $\mathfrak{p}$ is prime:

                    Suppose we have $xy\in \mathfrak{p},x,y\not\in \mathfrak{p}$.
                    Since $\mathfrak{p}$ is maximal and $\mathfrak{p}+(x),\mathfrak{p}+(y)\supsetneq \mathfrak{p}$,

                    $$

                    $$

                        (s_1-p_1)s_2&= a_1x(p_2+a_2y) \\
                        &= a_1p_2x+a_1a_2xy\in \mathfrak{p}
                    \end{align*}

            \end{enumerate}
            \qed

\begin{lemma}
    For any absolutely flat ring $A$ and $\forall a\in A,\exists x\in A,\text{s.t. }a=a^2x$.
\begin{proof}
    Consider any ideal $I,J\subseteq A$.
    Since $A$ is absolutely flat,
    $A / I$, as an $A$-module, is flat.

\begin{lemma}
    If the morphism $A\longrightarrow B$ is faithfully flat,
    then any finitely generated $A$-module $M,M'$ with sequence
    $$
    0\longrightarrow M'\longrightarrow M
    $$
    is exact iff
    $$
    0\longrightarrow M'\otimes _{A}B\longrightarrow M\otimes _{A}B
    $$
    is exact.
\begin{proof}
    For any $A$-module homomorphism $f:M'\longrightarrow M$, we have
    $$
    0\longrightarrow \ker{f}\longrightarrow M'\xrightarrow[]{f} M
    $$
    exact.
    By tensor product we have
    $$
    0\longrightarrow \ker{f}\otimes _{A}B\longrightarrow M'\otimes _{A}B\xrightarrow[]{f\otimes 1} M\otimes _{A}B
    $$
    exact.
    \begin{enumerate}[label=({\roman*})]
        \item When $0\longrightarrow \ker{f}\longrightarrow M'\xrightarrow[]{f} M$ is exact:

            Now $\ker{f}=0$, therefore
            $$
            \ker{f\otimes 1}=\ker{f}\otimes _{A}B=0
            .$$
            Done.
        \item When $0\longrightarrow \ker{f}\otimes _{A}B\longrightarrow M'\otimes _{A}B\xrightarrow[]{f\otimes 1} M\otimes _{A}B$ is exact:

            Now $\ker{f}\otimes _{A}B=0$.
            Denote the natural homomorphism $\phi:A\longrightarrow B$, and we have $B\cong A / \ker{\phi}$,
            \begin{align*}
                \ker{f}\otimes _{A}B
                &\cong \ker{f}\otimes _{A}\left( A / \ker{\phi} \right)  \\
                &= \ker{f} / \ker{\phi}\ker{f}
            \end{align*}
            i.e. $\ker{f} / \ker{f}\ker{\phi}=0$.
            By \emph{Proposition 3.8} we may suppose $A$ is local, at which $\ker{\phi}\subseteq \jab{A}$.
            Also, from $M,M'$ being finitely generated we know $\ker{f}$ is finitely generated.
            Hence, by \emph{Nakayama's Lemma} $\ker{f}=0$, i.e.
            $$
            0\longrightarrow M'\xrightarrow[]{f} M
            $$
            exact.
    \end{enumerate}
\end{proof}
\end{lemma}
